```python
from django.db import models


class ManicureDesign(models.Model):
    name = models.CharField(max_length=100, verbose_name="Име на дизайна")
    image = models.ImageField(
        upload_to="designs/",
        verbose_name="Изображение на дизайна"
    )
    created_at = models.DateTimeField(auto_now_add=True, verbose_name="Създадено на")

    class Meta:
        verbose_name = "Дизайн за маникюр"
        verbose_name_plural = "Дизайни за маникюр"

    def __str__(self):
        return self.name


class UserSession(models.Model):
    selected_design = models.ForeignKey(
        ManicureDesign,
        on_delete=models.SET_NULL,
        null=True,
        blank=True,
        related_name="user_sessions",
        verbose_name="Избран дизайн"
    )
    original_image = models.ImageField(
        upload_to="uploads/",
        verbose_name="Оригинална снимка"
    )
    processed_image = models.ImageField(
        upload_to="processed/",
        null=True,
        blank=True,
        verbose_name="Обработена снимка"
    )
    uploaded_at = models.DateTimeField(auto_now_add=True, verbose_name="Качено на")

    class Meta:
        verbose_name = "Потребителска сесия"
        verbose_name_plural = "Потребителски сесии"

    def __str__(self):
        return f"Сесия {self.id}"
```

```python
from django.shortcuts import get_object_or_404, render, redirect
from django.urls import reverse

from .models import ManicureDesign, UserSession


def index(request):
    designs = ManicureDesign.objects.all().order_by("-created_at")
    return render(request, "main/index.html", {"designs": designs})


def design_detail(request, design_id):
    design = get_object_or_404(ManicureDesign, pk=design_id)
    return render(request, "main/design_detail.html", {"design": design})


def upload_hand_image(request, design_id):
    design = get_object_or_404(ManicureDesign, pk=design_id)

    if request.method == "POST" and request.FILES.get("original_image"):
        user_session = UserSession.objects.create(
            selected_design=design,
            original_image=request.FILES["original_image"],
        )

        return redirect(reverse("main:index"))

    return render(request, "main/design_detail.html", {"design": design})
```

```python
from django.shortcuts import get_object_or_404, render, redirect
from django.urls import reverse

from .models import ManicureDesign, UserSession


def index(request):
    designs = ManicureDesign.objects.all().order_by("-created_at")
    return render(request, "main/index.html", {"designs": designs})


def design_detail(request, design_id):
    design = get_object_or_404(ManicureDesign, pk=design_id)
    return render(request, "main/design_detail.html", {"design": design})


def upload_hand_image(request, design_id):
    design = get_object_or_404(ManicureDesign, pk=design_id)

    if request.method == "POST" and request.FILES.get("original_image"):
        user_session = UserSession.objects.create(
            selected_design=design,
            original_image=request.FILES["original_image"],
        )

        return redirect(reverse("main:index"))

    return render(request, "main/design_detail.html", {"design": design})
```

```python
from django.shortcuts import get_object_or_404, render, redirect
from django.urls import reverse

from .forms import HandImageUploadForm
from .models import ManicureDesign, UserSession


def index(request):
    designs = ManicureDesign.objects.all().order_by("-created_at")
    return render(request, "main/index.html", {"designs": designs})


def design_detail(request, design_id):
    design = get_object_or_404(ManicureDesign, pk=design_id)
    form = HandImageUploadForm()
    return render(request, "main/design_detail.html", {"design": design, "form": form})


def upload_hand_image(request, design_id):
    design = get_object_or_404(ManicureDesign, pk=design_id)

    if request.method == "POST":
        form = HandImageUploadForm(request.POST, request.FILES)
        if form.is_valid():
            user_session = UserSession.objects.create(
                selected_design=design,
                original_image=form.cleaned_data["original_image"],
            )
            return redirect(reverse("main:result", kwargs={"session_id": user_session.id}))
    else:
        form = HandImageUploadForm()

    return render(
        request,
        "main/design_detail.html",
        {"design": design, "form": form}
    )


def result(request, session_id):
    user_session = get_object_or_404(UserSession, pk=session_id)
    return render(request, "main/result.html", {"session": user_session})
```

<llm-snippet-file>main/utils.py</llm-snippet-file>
```python
from pathlib import Path

from PIL import Image


def apply_design_overlay(hand_image_path, design_image_path, output_path, alpha=0.6):
    """
    Наслагва PNG дизайн върху снимка на ръка.

    Този първоначален вариант не прави откриване на нокти.
    Използва се като основа за бъдеща CV логика.
    """
    hand_image = Image.open(hand_image_path).convert("RGBA")
    design_image = Image.open(design_image_path).convert("RGBA")

    # Преоразмеряване на дизайна до размера на снимката
    design_image = design_image.resize(hand_image.size)

    # Регулиране на прозрачността на дизайна
    design_alpha = design_image.getchannel("A").point(lambda p: int(p * alpha))
    design_image.putalpha(design_alpha)

    # Комбиниране на двете изображения
    result = Image.alpha_composite(hand_image, design_image)

    # Уверяваме се, че изходната папка съществува
    output_path = Path(output_path)
    output_path.parent.mkdir(parents=True, exist_ok=True)

    # Запис на резултата
    result = result.convert("RGB")
    result.save(output_path)

    return output_path
```

## Какво прави тази функция
- отваря снимката на ръката;
- отваря PNG дизайна;
- мащабира дизайна;
- наслагва го с прозрачност;
- записва резултата във файл.

## Важно
Това е **начална версия**.  
Все още не:
- открива нокти;
- не позиционира дизайна по пръсти;
- не завърта модела според ръката.

Но е много полезна, защото ни дава **работеща основа**.

```python
from pathlib import Path

from PIL import Image


def apply_design_overlay(hand_image_path, design_image_path, output_path, alpha=0.7):
    """
    Наслагва PNG дизайн върху снимка на ръка.

    Този вариант е базов и служи като основа за бъдеща логика с
    откриване на нокти и по-прецизно позициониране.
    """
    hand_image = Image.open(hand_image_path).convert("RGBA")
    design_image = Image.open(design_image_path).convert("RGBA")

    hand_width, hand_height = hand_image.size

    # Запазваме пропорциите на дизайна и го мащабираме до по-малка част от снимката
    target_width = int(hand_width * 0.45)
    ratio = target_width / design_image.width
    target_height = int(design_image.height * ratio)

    design_image = design_image.resize((target_width, target_height), Image.Resampling.LANCZOS)

    # Регулиране на прозрачността
    if design_image.mode != "RGBA":
        design_image = design_image.convert("RGBA")

    alpha_channel = design_image.getchannel("A").point(lambda p: int(p * alpha))
    design_image.putalpha(alpha_channel)

    # Поставяне в центъра на изображението
    position = (
        (hand_width - design_image.width) // 2,
        (hand_height - design_image.height) // 2,
    )

    # Създаване на ново изображение и наслагване
    result = hand_image.copy()
    result.alpha_composite(design_image, dest=position)

    # Уверяваме се, че изходната папка съществува
    output_path = Path(output_path)
    output_path.parent.mkdir(parents=True, exist_ok=True)

    # Запис на резултата
    result = result.convert("RGB")
    result.save(output_path, quality=95)

    return output_path
```

```python
from pathlib import Path

import cv2
import mediapipe as mp
from PIL import Image


mp_hands = mp.solutions.hands
mp_drawing = mp.solutions.drawing_utils


def detect_hand_landmarks(image_path, output_path=None):
    """
    Открива ръка и ключови точки с MediaPipe Hands.
    По желание записва изображение с визуализирани landmarks.
    """
    image_bgr = cv2.imread(str(image_path))

    if image_bgr is None:
        raise ValueError("Неуспешно зареждане на изображението.")

    image_rgb = cv2.cvtColor(image_bgr, cv2.COLOR_BGR2RGB)

    with mp_hands.Hands(
        static_image_mode=True,
        max_num_hands=1,
        min_detection_confidence=0.5
    ) as hands:
        results = hands.process(image_rgb)

        if not results.multi_hand_landmarks:
            return {
                "found": False,
                "landmarks": None,
                "output_path": None,
            }

        hand_landmarks = results.multi_hand_landmarks[0]

        if output_path:
            annotated_image = image_rgb.copy()
            mp_drawing.draw_landmarks(
                annotated_image,
                hand_landmarks,
                mp_hands.HAND_CONNECTIONS
            )

            annotated_bgr = cv2.cvtColor(annotated_image, cv2.COLOR_RGB2BGR)
            output_path = Path(output_path)
            output_path.parent.mkdir(parents=True, exist_ok=True)
            cv2.imwrite(str(output_path), annotated_bgr)

        landmarks = []
        height, width, _ = image_rgb.shape

        for lm in hand_landmarks.landmark:
            x = int(lm.x * width)
            y = int(lm.y * height)
            landmarks.append((x, y))

        return {
            "found": True,
            "landmarks": landmarks,
            "output_path": str(output_path) if output_path else None,
        }


def apply_design_overlay(hand_image_path, design_image_path, output_path, alpha=0.7):
    """
    Наслагва PNG дизайн върху снимка на ръка.

    Този вариант е базов и служи като основа за бъдеща логика с
    откриване на нокти и по-прецизно позициониране.
    """
    hand_image = Image.open(hand_image_path).convert("RGBA")
    design_image = Image.open(design_image_path).convert("RGBA")

    hand_width, hand_height = hand_image.size

    target_width = int(hand_width * 0.45)
    ratio = target_width / design_image.width
    target_height = int(design_image.height * ratio)

    design_image = design_image.resize(
        (target_width, target_height),
        Image.Resampling.LANCZOS
    )

    if design_image.mode != "RGBA":
        design_image = design_image.convert("RGBA")

    alpha_channel = design_image.getchannel("A").point(lambda p: int(p * alpha))
    design_image.putalpha(alpha_channel)

    position = (
        (hand_width - design_image.width) // 2,
        (hand_height - design_image.height) // 2,
    )

    result = hand_image.copy()
    result.alpha_composite(design_image, dest=position)

    output_path = Path(output_path)
    output_path.parent.mkdir(parents=True, exist_ok=True)

    result = result.convert("RGB")
    result.save(output_path, quality=95)

    return output_path


def process_hand_image_with_landmarks(hand_image_path, output_path):
    """
    Зарежда изображение, открива landmarks и записва визуализирана версия.
    """
    return detect_hand_landmarks(hand_image_path, output_path=output_path)
```

```python
from pathlib import Path
import math

import cv2
import mediapipe as mp
from PIL import Image


mp_hands = mp.solutions.hands
mp_drawing = mp.solutions.drawing_utils


FINGER_TIPS = {
    "thumb": 4,
    "index": 8,
    "middle": 12,
    "ring": 16,
    "pinky": 20,
}

FINGER_DIPS = {
    "thumb": 3,
    "index": 7,
    "middle": 11,
    "ring": 15,
    "pinky": 19,
}


def detect_hand_landmarks(image_path, output_path=None):
    """
    Открива ръка и ключови точки с MediaPipe Hands.
    По желание записва изображение с визуализирани landmarks.
    """
    image_bgr = cv2.imread(str(image_path))

    if image_bgr is None:
        raise ValueError("Неуспешно зареждане на изображението.")

    image_rgb = cv2.cvtColor(image_bgr, cv2.COLOR_BGR2RGB)

    with mp_hands.Hands(
        static_image_mode=True,
        max_num_hands=1,
        min_detection_confidence=0.5
    ) as hands:
        results = hands.process(image_rgb)

        if not results.multi_hand_landmarks:
            return {
                "found": False,
                "landmarks": None,
                "output_path": None,
            }

        hand_landmarks = results.multi_hand_landmarks[0]

        if output_path:
            annotated_image = image_rgb.copy()
            mp_drawing.draw_landmarks(
                annotated_image,
                hand_landmarks,
                mp_hands.HAND_CONNECTIONS
            )

            annotated_bgr = cv2.cvtColor(annotated_image, cv2.COLOR_RGB2BGR)
            output_path = Path(output_path)
            output_path.parent.mkdir(parents=True, exist_ok=True)
            cv2.imwrite(str(output_path), annotated_bgr)

        landmarks = []
        height, width, _ = image_rgb.shape

        for lm in hand_landmarks.landmark:
            x = int(lm.x * width)
            y = int(lm.y * height)
            landmarks.append((x, y))

        return {
            "found": True,
            "landmarks": landmarks,
            "output_path": str(output_path) if output_path else None,
        }


def get_nail_regions(landmarks):
    """
    Изчислява приблизителни зони за ноктите на база landmarks.

    Връща речник с център, ширина, височина и ъгъл за всеки пръст.
    """
    if not landmarks or len(landmarks) < 21:
        return {}

    nail_regions = {}

    for finger_name, tip_idx in FINGER_TIPS.items():
        dip_idx = FINGER_DIPS[finger_name]

        tip_x, tip_y = landmarks[tip_idx]
        dip_x, dip_y = landmarks[dip_idx]

        # Вектор от DIP към TIP
        dx = tip_x - dip_x
        dy = tip_y - dip_y

        # Центърът на нокътната зона е малко под върха на пръста
        center_x = int(tip_x + dx * 0.25)
        center_y = int(tip_y + dy * 0.25)

        # Размерът на зоната се базира на разстоянието между TIP и DIP
        distance = math.sqrt(dx * dx + dy * dy)
        nail_width = max(18, int(distance * 0.55))
        nail_height = max(28, int(distance * 0.85))

        # Ъгълът на пръста
        angle = math.degrees(math.atan2(dy, dx))

        nail_regions[finger_name] = {
            "center": (center_x, center_y),
            "width": nail_width,
            "height": nail_height,
            "angle": angle,
            "tip": (tip_x, tip_y),
            "dip": (dip_x, dip_y),
        }

    return nail_regions


def detect_hand_landmarks_and_nails(image_path, output_path=None):
    """
    Открива ръка и връща landmarks + приблизителни нокътни зони.
    """
    detection = detect_hand_landmarks(image_path, output_path=output_path)

    if not detection["found"]:
        return {
            "found": False,
            "landmarks": None,
            "nail_regions": {},
            "output_path": detection["output_path"],
        }

    nail_regions = get_nail_regions(detection["landmarks"])

    return {
        "found": True,
        "landmarks": detection["landmarks"],
        "nail_regions": nail_regions,
        "output_path": detection["output_path"],
    }


def apply_design_overlay(hand_image_path, design_image_path, output_path, alpha=0.7):
    """
    Наслагва PNG дизайн върху снимка на ръка.

    Този вариант е базов и служи като основа за бъдеща логика с
    откриване на нокти и по-прецизно позициониране.
    """
    hand_image = Image.open(hand_image_path).convert("RGBA")
    design_image = Image.open(design_image_path).convert("RGBA")

    hand_width, hand_height = hand_image.size

    target_width = int(hand_width * 0.45)
    ratio = target_width / design_image.width
    target_height = int(design_image.height * ratio)

    design_image = design_image.resize(
        (target_width, target_height),
        Image.Resampling.LANCZOS
    )

    if design_image.mode != "RGBA":
        design_image = design_image.convert("RGBA")

    alpha_channel = design_image.getchannel("A").point(lambda p: int(p * alpha))
    design_image.putalpha(alpha_channel)

    position = (
        (hand_width - design_image.width) // 2,
        (hand_height - design_image.height) // 2,
    )

    result = hand_image.copy()
    result.alpha_composite(design_image, dest=position)

    output_path = Path(output_path)
    output_path.parent.mkdir(parents=True, exist_ok=True)

    result = result.convert("RGB")
    result.save(output_path, quality=95)

    return output_path
```

## Какво добавихме
### `FINGER_TIPS` и `FINGER_DIPS`
Това са индексите на върховете и междинните стави на пръстите:
- палец
- показалец
- среден
- безименен
- кутре

### `get_nail_regions(landmarks)`
Тази функция:
- взема върха и долната става на всеки пръст;
- изчислява:
  - център на нокътя;
  - ширина;
  - височина;
  - ъгъл на пръста.

### `detect_hand_landmarks_and_nails(image_path)`
Комбинира:
- детекция на ръката;
- изчисляване на нокътни зони.

Следващата логична стъпка е:
- да вземем `nail_regions`;

- вместо да слагаме един дизайн върху цялата снимка, да го поставяме върху всеки нокът.

````plain text
from pathlib import Path
import math

import cv2
import mediapipe as mp
from PIL import Image


try:
    mp_hands = mp.solutions.hands
    mp_drawing = mp.solutions.drawing_utils
except AttributeError:
    try:
        from mediapipe.python.solutions import hands as mp_hands
        from mediapipe.python.solutions import drawing_utils as mp_drawing
    except ImportError as exc:
        raise ImportError(
            "Не може да се зареди MediaPipe Hands API. "
            "Провери инсталираната версия на mediapipe, защото наличният пакет "
            "не предоставя expected solutions API."
        ) from exc


FINGER_TIPS = {
    "thumb": 4,
    "index": 8,
    "middle": 12,
    "ring": 16,
    "pinky": 20,
}

FINGER_DIPS = {
    "thumb": 3,
    "index": 7,
    "middle": 11,
    "ring": 15,
    "pinky": 19,
}


def detect_hand_landmarks(image_path, output_path=None):
    """
    Открива ръка и ключови точки с MediaPipe Hands.
    По желание записва изображение с визуализирани landmarks.
    """
    image_bgr = cv2.imread(str(image_path))

    if image_bgr is None:
        raise ValueError("Неуспешно зареждане на изображението.")

    image_rgb = cv2.cvtColor(image_bgr, cv2.COLOR_BGR2RGB)

    with mp_hands.Hands(
        static_image_mode=True,
        max_num_hands=1,
        min_detection_confidence=0.5
    ) as hands:
        results = hands.process(image_rgb)

        if not results.multi_hand_landmarks:
            return {
                "found": False,
                "landmarks": None,
                "output_path": None,
            }

        hand_landmarks = results.multi_hand_landmarks[0]

        if output_path:
            annotated_image = image_rgb.copy()
            mp_drawing.draw_landmarks(
                annotated_image,
                hand_landmarks,
                mp_hands.HAND_CONNECTIONS
            )

            annotated_bgr = cv2.cvtColor(annotated_image, cv2.COLOR_RGB2BGR)
            output_path = Path(output_path)
            output_path.parent.mkdir(parents=True, exist_ok=True)
            cv2.imwrite(str(output_path), annotated_bgr)

        landmarks = []
        height, width, _ = image_rgb.shape

        for lm in hand_landmarks.landmark:
            x = int(lm.x * width)
            y = int(lm.y * height)
            landmarks.append((x, y))

        return {
            "found": True,
            "landmarks": landmarks,
            "output_path": str(output_path) if output_path else None,
        }


def get_nail_regions(landmarks):
    """
    Изчислява приблизителни зони за ноктите на база landmarks.

    Връща речник с център, ширина, височина и ъгъл за всеки пръст.
    """
    if not landmarks or len(landmarks) < 21:
        return {}

    nail_regions = {}

    for finger_name, tip_idx in FINGER_TIPS.items():
        dip_idx = FINGER_DIPS[finger_name]

        tip_x, tip_y = landmarks[tip_idx]
        dip_x, dip_y = landmarks[dip_idx]

        # Вектор от DIP към TIP
        dx = tip_x - dip_x
        dy = tip_y - dip_y

        # Центърът на нокътната зона е малко под върха на пръста
        center_x = int(tip_x + dx * 0.25)
        center_y = int(tip_y + dy * 0.25)

        # Размерът на зоната се базира на разстоянието между TIP и DIP
        distance = math.sqrt(dx * dx + dy * dy)
        nail_width = max(18, int(distance * 0.55))
        nail_height = max(28, int(distance * 0.85))

        # Ъгълът на пръста
        angle = math.degrees(math.atan2(dy, dx))

        nail_regions[finger_name] = {
            "center": (center_x, center_y),
            "width": nail_width,
            "height": nail_height,
            "angle": angle,
            "tip": (tip_x, tip_y),
            "dip": (dip_x, dip_y),
        }

    return nail_regions


def detect_hand_landmarks_and_nails(image_path, output_path=None):
    """
    Открива ръка и връща landmarks + приблизителни нокътни зони.
    """
    detection = detect_hand_landmarks(image_path, output_path=output_path)

    if not detection["found"]:
        return {
            "found": False,
            "landmarks": None,
            "nail_regions": {},
            "output_path": detection["output_path"],
        }

    nail_regions = get_nail_regions(detection["landmarks"])

    return {
        "found": True,
        "landmarks": detection["landmarks"],
        "nail_regions": nail_regions,
        "output_path": detection["output_path"],
    }


def apply_design_overlay(hand_image_path, design_image_path, output_path, alpha=0.7):
    """
    Наслагва PNG дизайн върху снимка на ръка.

    Този вариант е базов и служи като основа за бъдеща логика с
    откриване на нокти и по-прецизно позициониране.
    """
    hand_image = Image.open(hand_image_path).convert("RGBA")
    design_image = Image.open(design_image_path).convert("RGBA")

    hand_width, hand_height = hand_image.size

    target_width = int(hand_width * 0.45)
    ratio = target_width / design_image.width
    target_height = int(design_image.height * ratio)

    design_image = design_image.resize(
        (target_width, target_height),
        Image.Resampling.LANCZOS
    )

    if design_image.mode != "RGBA":
        design_image = design_image.convert("RGBA")

    alpha_channel = design_image.getchannel("A").point(lambda p: int(p * alpha))
    design_image.putalpha(alpha_channel)

    position = (
        (hand_width - design_image.width) // 2,
        (hand_height - design_image.height) // 2,
    )

    result = hand_image.copy()
    result.alpha_composite(design_image, dest=position)

    output_path = Path(output_path)
    output_path.parent.mkdir(parents=True, exist_ok=True)

    result = result.convert("RGB")
    result.save(output_path, quality=95)

    return output_path


def process_hand_image_with_landmarks(hand_image_path, output_path):
    """
    Зарежда изображение, открива landmarks и записва визуализирана версия.
    """
    return detect_hand_landmarks(hand_image_path, output_path=output_path)

## Какъв е проблемът

Грешката не е в Django URL-ите, а при **импортирането на `main.utils`**:

- `views.py` зарежда `apply_design_overlay`
- това импортва `main/utils.py`
- в `utils.py` има **глобален импорт/проверка за MediaPipe Hands API**
- при вашата инсталация `mediapipe` **няма очаквания `solutions` API**, затова модулът хвърля `ImportError`
- резултатът е, че **сървърът не стартира**, въпреки че реално страницата за наслагване може да работи и без Hand landmarks

## Защо това е лошо решение в момента

`apply_design_overlay()` **не използва MediaPipe**.  
Затова е грешно целият модул да спира зареждането само заради функционалност, която е второстепенна.

## Дългосрочно решение

Най-добре е:

1. **Да махнете hard-fail импорта** на MediaPipe от ниво модул.
2. **Да зареждате MediaPipe лениво** само във функциите, които наистина го ползват.
3. Ако MediaPipe липсва или е несъвместим, да:
   - връщате ясна грешка само при използване на landmark функциите
   - но **да не блокирате целия сайт**

Така приложението ще стартира нормално и базовото наслагване ще работи.

    with mp_hands.Hands(
        static_image_mode=True,
        max_num_hands=1,
        min_detection_confidence=0.5
    ) as hands:
        results = hands.process(image_rgb)

        if not results.multi_hand_landmarks:
            return {
                "found": False,
                "landmarks": None,
                "output_path": None,
            }

        hand_landmarks = results.multi_hand_landmarks[0]

        if output_path:
            annotated_image = image_rgb.copy()
            mp_drawing.draw_landmarks(
                annotated_image,
                hand_landmarks,
                mp_hands.HAND_CONNECTIONS
            )

            annotated_bgr = cv2.cvtColor(annotated_image, cv2.COLOR_RGB2BGR)
            output_path = Path(output_path)
            output_path.parent.mkdir(parents=True, exist_ok=True)
            cv2.imwrite(str(output_path), annotated_bgr)

        landmarks = []
        height, width, _ = image_rgb.shape

        for lm in hand_landmarks.landmark:
            x = int(lm.x * width)
            y = int(lm.y * height)
            landmarks.append((x, y))

        return {
            "found": True,
            "landmarks": landmarks,
            "output_path": str(output_path) if output_path else None,
        }


def get_nail_regions(landmarks):
    """
    Изчислява приблизителни зони за ноктите на база landmarks.

    Връща речник с център, ширина, височина и ъгъл за всеки пръст.
    """
    if not landmarks or len(landmarks) < 21:
        return {}

    nail_regions = {}

    for finger_name, tip_idx in FINGER_TIPS.items():
        dip_idx = FINGER_DIPS[finger_name]

        tip_x, tip_y = landmarks[tip_idx]
        dip_x, dip_y = landmarks[dip_idx]

        dx = tip_x - dip_x
        dy = tip_y - dip_y

        center_x = int(tip_x + dx * 0.25)
        center_y = int(tip_y + dy * 0.25)

        distance = math.sqrt(dx * dx + dy * dy)
        nail_width = max(18, int(distance * 0.55))
        nail_height = max(28, int(distance * 0.85))

        angle = math.degrees(math.atan2(dy, dx))

        nail_regions[finger_name] = {
            "center": (center_x, center_y),
            "width": nail_width,
            "height": nail_height,
            "angle": angle,
            "tip": (tip_x, tip_y),
            "dip": (dip_x, dip_y),
        }

    return nail_regions


def detect_hand_landmarks_and_nails(image_path, output_path=None):
    """
    Открива ръка и връща landmarks + приблизителни нокътни зони.
    """
    detection = detect_hand_landmarks(image_path, output_path=output_path)

    if not detection["found"]:
        return {
            "found": False,
            "landmarks": None,
            "nail_regions": {},
            "output_path": detection["output_path"],
        }

    nail_regions = get_nail_regions(detection["landmarks"])

    return {
        "found": True,
        "landmarks": detection["landmarks"],
        "nail_regions": nail_regions,
        "output_path": detection["output_path"],
    }


def apply_design_overlay(hand_image_path, design_image_path, output_path, alpha=0.7):
    """
    Наслагва PNG дизайн върху снимка на ръка.

    Този вариант е базов и служи като основа за бъдеща логика с
    откриване на нокти и по-прецизно позициониране.
    """
    hand_image = Image.open(hand_image_path).convert("RGBA")
    design_image = Image.open(design_image_path).convert("RGBA")

    hand_width, hand_height = hand_image.size

    target_width = int(hand_width * 0.45)
    ratio = target_width / design_image.width
    target_height = int(design_image.height * ratio)

    design_image = design_image.resize(
        (target_width, target_height),
        Image.Resampling.LANCZOS
    )

    if design_image.mode != "RGBA":
        design_image = design_image.convert("RGBA")

    alpha_channel = design_image.getchannel("A").point(lambda p: int(p * alpha))
    design_image.putalpha(alpha_channel)

    position = (
        (hand_width - design_image.width) // 2,
        (hand_height - design_image.height) // 2,
    )

    result = hand_image.copy()
    result.alpha_composite(design_image, dest=position)

    output_path = Path(output_path)
    output_path.parent.mkdir(parents=True, exist_ok=True)

    result = result.convert("RGB")
    result.save(output_path, quality=95)

    return output_path


def process_hand_image_with_landmarks(hand_image_path, output_path):
    """
    Зарежда изображение, открива landmarks и записва визуализирана версия.
    """
    return detect_hand_landmarks(hand_image_path, output_path=output_path)
````
